<script>
(function(){
  const canvas = document.getElementById('confettiCanvas');
  const ctx = canvas.getContext('2d');
  let W = 0, H = 0, confetti = [], animationId = null;
  const MAX_CONFETTI = 600; // total cap
  const SMALL_SCREEN_CAP = 120;

  function resizeCanvas(){
    const rect = canvas.getBoundingClientRect();
    const dpr = window.devicePixelRatio || 1;
    canvas.width = rect.width * dpr;
    canvas.height = rect.height * dpr;
    canvas.style.width = rect.width + "px";
    canvas.style.height = rect.height + "px";
    ctx.setTransform(dpr,0,0,dpr,0,0);
    W = rect.width; H = rect.height;
  }

  function rand(min,max){ return Math.random()*(max-min)+min; }

  function spawnConfetti(count = 280){
    // do not exceed MAX_CONFETTI
    const allowable = Math.max(0, MAX_CONFETTI - confetti.length);
    if (allowable <= 0) return;
    let desired = Math.min(count, allowable);
    const colors=['#ff2d7a','#fff176','#8bc34a','#00bcd4','#ff8bcf'];
    const limit = Math.min(desired, window.innerWidth < 480 ? SMALL_SCREEN_CAP : desired);
    for(let i=0;i<limit;i++){
      confetti.push({
        x: rand(0,W), y: rand(-H,0), s: rand(6,12),
        vx: rand(-1.5,1.5), vy: rand(2,6),
        c: colors[Math.floor(rand(0,colors.length))],
        rot: rand(0,360), vr: rand(-0.1,0.1)
      });
    }
    if(!animationId) animate();
  }

  function drawConfetti(){
    ctx.clearRect(0,0,W,H);
    for (let i = confetti.length - 1; i >= 0; i--){
      const p = confetti[i];
      p.x += p.vx; p.y += p.vy; p.rot += p.vr;
      ctx.save();
      ctx.translate(p.x + p.s/2, p.y + (p.s*0.3));
      ctx.rotate((p.rot * Math.PI)/180);
      ctx.fillStyle = p.c;
      ctx.fillRect(-p.s/2, -p.s*0.3, p.s, p.s*0.6);
      ctx.restore();
    }
    // remove offscreen
    confetti = confetti.filter(p => p.y < H + 40);
    if (confetti.length === 0){
      cancelAnimationFrame(animationId);
      animationId = null;
    }
  }

  function animate(){
    animationId = requestAnimationFrame(animate);
    drawConfetti();
  }

  /* MEMORY popup */
  function showMemoryPopup(){
    if(document.getElementById('memoryPopupOverlay')) return;

    const overlay = document.createElement('div');
    overlay.id = 'memoryPopupOverlay';
    overlay.style.position = 'fixed';
    overlay.style.left = 0; overlay.style.top = 0; overlay.style.right = 0; overlay.style.bottom = 0;
    overlay.style.background = 'rgba(0,0,0,0.45)';
    overlay.style.display = 'flex'; overlay.style.alignItems = 'center'; overlay.style.justifyContent = 'center';
    overlay.style.zIndex = 99999;

    const box = document.createElement('div');
    box.setAttribute('role','dialog');
    box.setAttribute('aria-modal','true');
    box.style.width = 'min(500px,90%)';
    box.style.background = 'linear-gradient(180deg,#fff,#fffaf7)';
    box.style.borderRadius = '20px';
    box.style.padding = '20px 22px';
    box.style.boxShadow = '0 20px 60px rgba(0,0,0,0.2)';
    box.style.textAlign = 'left';

    const title = document.createElement('div');
    title.className = 'popup-title';
    title.textContent = 'Happy Birthday My Dear Princess 🎀🎂';

    const img = document.createElement('img');
    img.className = 'popup-img';
    // If you want to use a different image, replace this URL or host it with your site.
    img.src = 'https://i.ibb.co/ZRybyWQ5/IMG-20251120-000301.jpg';
    img.alt = 'Princee photo';
    img.loading = 'lazy';

    const note = document.createElement('p');
    note.textContent = 'Chhipakalee 😃 Just kidding 😃';

    const close = document.createElement('button');
    close.textContent = 'Close';
    close.style.marginTop = '12px';
    close.style.padding = '10px 14px';
    close.style.borderRadius = '12px';
    close.style.border = '0';
    close.style.background = 'linear-gradient(90deg,#ff2d7a,#ff83c7)';
    close.style.color = 'white';
    close.style.fontWeight = '700';
    close.style.cursor = 'pointer';
    close.onclick = () => overlay.remove();
    close.onkeydown = e => { if (e.key === 'Enter' || e.key === ' ') { overlay.remove(); } };

    box.appendChild(title);
    box.appendChild(img);
    box.appendChild(note);
    box.appendChild(close);
    overlay.appendChild(box);

    overlay.addEventListener('click', e => { if (e.target === overlay) overlay.remove(); });

    document.body.appendChild(overlay);
    close.focus();
  }

  function attachHandlers(){
    const confettiBtn = document.getElementById('confettiBtn');
    const msgBtn = document.getElementById('msgBtn');
    const song = document.getElementById('bgSong');

    confettiBtn.addEventListener('click', () => {
      spawnConfetti(280);
      if (song){
        try {
          // set a reasonable starting volume
          song.volume = 0.8;
          // play, but ignore promise rejection when autoplay blocked — user gesture clicked so usually succeeds
          song.play().catch(()=>{ /* silence */ });
        } catch(e){}
      }
    });

    // keyboard accessibility (Enter/Space) already native on <button>, but ensure message button opens popup when activated by keyboard
    msgBtn.addEventListener('click', showMemoryPopup);

    // window resizing
    window.addEventListener('resize', () => setTimeout(resizeCanvas, 80));
  }

  // init
  window.addEventListener('load', () => { resizeCanvas(); attachHandlers(); });
})();
</script>
